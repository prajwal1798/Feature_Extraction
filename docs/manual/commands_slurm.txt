# // ==========================================================================
# // COMMANDS FOR A SLURM CLUSTER (written on Sunbird, Supercomputing Wales),
# // IN THE ORDER OF USE
# //
# // Log in from your computer first:
# //     ssh USERNAME@LOGIN_NODE
# // USERNAME / LOGIN_NODE: your cluster user name and login address.
# // Before the first job, edit the cluster lines at the top of
# // jobs/slurm/gpu.sh and jobs/slurm/cpu.sh (account, partition, GPU type).
# // Paths use /scratch/$USER; if your cluster's work disk is elsewhere,
# // replace /scratch/$USER by it throughout.
# //
# // EVERYTHING lives in ONE folder:  /scratch/$USER/ML
# //     ML/run.py, ML/src, ML/jobs   the code
# //     ML/env                       the Python environment
# //     ML/data                      the datasets (made by the code)
# //     ML/outputs                   every result (made by the code)
# // Every sbatch line must be typed inside ML:   cd /scratch/$USER/ML
# // Lines starting with "#" are notes: bash ignores them, so pasting is safe.
# // A line ending in \ continues on the next line: copy both together.
# //
# // Pattern of every project command:
# //     sbatch jobs/slurm/gpu.sh  STAGE  --option value ...   (networks, A100)
# //     sbatch jobs/slurm/cpu.sh  STAGE  --option value ...   (everything else)
# // The words after gpu.sh / cpu.sh are those you would type after
# // "python run.py" on your own computer.
# //
# // Submit one block, wait until "squeue -u $USER" shows nothing, then submit
# // the next block: each step needs the results of the one before.
# //
# // HOW TO SEE A JOB                                           block  0
# // PART A  set up (once)                                      blocks 1 - 2
# // PART B  THE FULL STUDY, step by step                       blocks 3 - 8
# // PART C  the earlier 500-case study                         blocks 9 - 11
# // PART D  stopping, results                                  blocks 12 - 14
# // ==========================================================================


# ##########################################################################
# #  0.  HOW TO SEE A JOB  (read once; used after every sbatch below)
# ##########################################################################
# //
# // sbatch answers:       Submitted batch job 8206449
# // 8206449 is the job number. Everything the job prints goes into ONE file:
# //       outputs/slurm/gpu_8206449.out    (jobs sent with gpu.sh)
# //       outputs/slurm/cpu_8206449.out    (jobs sent with cpu.sh)
# // The file starts with JOB / NODE / STARTED / COMMAND and ends with
# // FINISHED and EXIT CODE (0 = success).
# //
# // squeue -u $USER     : my jobs. ST column: PD waiting, R running,
# //                       CG finishing. A job missing from the list is over.
# // tail -F FILE        : the output LIVE, line by line as it is printed.
# //                       Waits if the job has not started yet.
# //                       Ctrl+C stops WATCHING; the job keeps running.
# // cat FILE            : the whole output of a finished job
# // tail -n 30 FILE     : its last 30 lines (errors are at the end)
# // ls -t outputs/slurm | head   : the newest output files, newest first
# // sacct -j NUMBER     : COMPLETED / FAILED / TIMEOUT / CANCELLED
# // scancel NUMBER      : stop a job
# //
# // ARRAY JOBS (--array=0-24): every array line below also has
# //       --output=outputs/slurm/gpu_%A_%a.out
# // so task 3 of array 8206500 writes  outputs/slurm/gpu_8206500_3.out
# //       tail -F outputs/slurm/gpu_8206500_3.out
# // (squeue shows it as 8206500_3; srun --jobid=8206500_3 --overlap nvidia-smi
# //  shows its GPU). Arrays submitted without --output: find a task's file with
# //       grep -l "array 8206500 task 3)" outputs/slurm/*.out
# //
# // The RESULTS (tables, figures, models) are not in outputs/slurm; each
# // block below says where they go ("see:"). All are under ML/outputs and
# // ML/data, and come home with block 14.
# // ==========================================================================
squeue -u $USER
tail -F outputs/slurm/cpu_8206449.out
cat outputs/slurm/cpu_8206449.out
tail -n 30 outputs/slurm/cpu_8206449.out
ls -t outputs/slurm | head
sacct -j 8206449


# ##########################################################################
# #  PART A   SET UP (ONCE)
# ##########################################################################

# // ==========================================================================
# // 1. GET THE PROJECT AND MAKE THE ENVIRONMENT
# //
# // cd /scratch/$USER      : your fast working disk
# // git clone ... ML       : downloads the project into a folder called ML
# //                          
# // mkdir -p outputs/slurm : folder for the job output files (jobs fail without it)
# // module load / source   : make the "conda" command available (the module
# //                          name differs between clusters: module avail conda)
# // conda create --prefix  : Python 3.11 in ML/env (the path the jobs activate)
# // pip install            : the four packages (on Linux, torch includes CUDA),
# //                          called through the environment's own Python so they
# //                          cannot land anywhere else. --only-binary=:all: takes
# //                          ready-made packages only: some clusters' system compiler
# //                          is too old to build NumPy from source.
# // ==========================================================================
cd /scratch/$USER
git clone https://github.com/prajwal1798/Feature_Extraction.git ML
cd ML
mkdir -p outputs/slurm
module load anaconda/2024.06
source "$(conda info --base)/etc/profile.d/conda.sh"
conda create --prefix /scratch/$USER/ML/env python=3.11 -y \
             --override-channels -c conda-forge
conda activate /scratch/$USER/ML/env
/scratch/$USER/ML/env/bin/python -m pip install --no-cache-dir --only-binary=:all: \
       numpy matplotlib ngsolve torch

# // ==========================================================================
# // 2. CHECK
# //
# // gpu-info : the output file must name an A100
# // smoke    : every stage on 20 tiny cases; must end with "Smoke test passed"
# // Every later login: cd /scratch/$USER/ML
# // ==========================================================================
sbatch jobs/slurm/gpu.sh gpu-info
sbatch jobs/slurm/cpu.sh smoke
# // see: gpu-info -> outputs/slurm/gpu_NUMBER.out shows the GPU (A100) in a few
# //                 seconds; the same record in
# //                 outputs/study_results/combined_0500/hardware_preflight.json
# //      smoke    -> tail -F outputs/slurm/cpu_NUMBER.out ; last lines must be
# //                 "Smoke test passed" and "EXIT CODE: 0"; its files are in
# //                 verification/ (can be deleted afterwards)


# ##########################################################################
# #  PART B   THE FULL STUDY, STEP BY STEP
# #
# #  One dataset ("scaling", 3800 FE cases) serves the whole study:
# #    3200 TRAINING cases (Halton sequence); training set N = first N cases,
# #         N = 200, 400, 800, 1600, 3200
# #     200 VALIDATION cases - the same for every N and every seed
# #     400 TEST cases       - the same for every N and every seed
# #  "Scaling" = how the test error scales with N:  error = C N^(-alpha)
# ##########################################################################

# // ==========================================================================
# // 3. DATA  (CPU node; if the 24-hour limit stops it, submit it again)
# // ==========================================================================
sbatch jobs/slurm/cpu.sh generate --family scaling
# // see: tail -F outputs/slurm/cpu_NUMBER.out  (one line per solved case)
# //      dataset -> data/processed/scaling_3200/protocol_v2/dataset.npz

# // when it has finished:
# // generate-rounded : 50 rounded rectangles (a shape never trained on)
# // plot-sampling    : figure of where the cavities lie
# // mesh-check       : FE accuracy on three meshes + heat balance
sbatch jobs/slurm/cpu.sh generate-rounded --family scaling --rounded-cases 50
sbatch jobs/slurm/cpu.sh plot-sampling --family scaling
sbatch jobs/slurm/cpu.sh mesh-check
# // see: rounded       -> data/external/scaling_3200/protocol_v2/rounded_rect_external_test/
# //      plot-sampling -> outputs/figures/scaling_3200/dataset/*.pdf, *.png
# //      mesh-check    -> outputs/study_results/combined_0500/mesh_convergence/

# // ==========================================================================
# // 4. QUICK TRY OF ONE MODEL  (optional)
# //
# // F="..." stores the common words; $F pastes them back.
# // ==========================================================================
F="--family scaling --operator fno"
sbatch jobs/slurm/gpu.sh train $F --train-size 800 --seed 321 --run-name try_fno --epochs 300
# // see: tail -F outputs/slurm/gpu_NUMBER.out  (one line every 20 epochs:
# //      training MSE, validation MSE, RMSE in K)
# //      loss history -> outputs/logs/scaling_3200/try_fno/loss_history.csv
# //      model        -> outputs/checkpoints/scaling_3200/try_fno/best_model.pt

# // ==========================================================================
# // 5. TUNING AT N = 800  (validation cases only)
# //
# // --array=0-24 : 25 tasks at the same time, task k trains model k
# // tune-select  : submit only when the array before it has finished
# // Keep the order: each stage starts from the winner of the one before.
# // Set T again after a new login.
# // ==========================================================================
T="--family scaling --train-size 800 --operator deeponet"
sbatch --array=0-24 --output=outputs/slurm/gpu_%A_%a.out jobs/slurm/gpu.sh tune $T --stage-name grid --epochs 2000
sbatch jobs/slurm/cpu.sh tune-select $T --stage-name grid
sbatch --array=0-24 --output=outputs/slurm/gpu_%A_%a.out jobs/slurm/gpu.sh tune $T --stage-name grid_gelu --epochs 2000
sbatch jobs/slurm/cpu.sh tune-select $T --stage-name grid_gelu
sbatch --array=0-2 --output=outputs/slurm/gpu_%A_%a.out jobs/slurm/gpu.sh tune $T --stage-name lr --epochs 2000
sbatch jobs/slurm/cpu.sh tune-select $T --stage-name lr

T="--family scaling --train-size 800 --operator fno"
sbatch --array=0-11 --output=outputs/slurm/gpu_%A_%a.out jobs/slurm/gpu.sh tune $T --stage-name modes_width --epochs 2000
sbatch jobs/slurm/cpu.sh tune-select $T --stage-name modes_width
sbatch --array=0-3 --output=outputs/slurm/gpu_%A_%a.out jobs/slurm/gpu.sh tune $T --stage-name depth --epochs 2000
sbatch jobs/slurm/cpu.sh tune-select $T --stage-name depth
sbatch --array=0-2 --output=outputs/slurm/gpu_%A_%a.out jobs/slurm/gpu.sh tune $T --stage-name lr --epochs 2000
sbatch jobs/slurm/cpu.sh tune-select $T --stage-name lr
# // see: tune        -> one output file per task (block 0, ARRAY JOBS);
# //                    squeue shows how many tasks are still running
# //      tune-select -> cpu_NUMBER.out prints the table and the winner;
# //                    tables     outputs/study_results/scaling_3200/tuning/*.csv
# //                    winner     outputs/study_results/scaling_3200/tuning/*_selected.json
# //                    heatmaps   outputs/figures/scaling_3200/report/N0800/tuning_*.pdf

# // ==========================================================================
# // 6. THE DATA-SIZE RUNS  (5 sizes x 5 seeds = 25 tasks per operator)
# //
# // task k = size number k/5 (200 ... 3200), seed number k%5
# //          (321, 777, 2026, 31415, 27182); runs scale_fno_N0800_seed321, ...
# // If the 24-hour limit stops large runs, submit the same lines again:
# // finished runs are skipped, interrupted ones continue.
# // ==========================================================================
sbatch --array=0-24 --output=outputs/slurm/gpu_%A_%a.out jobs/slurm/gpu.sh train-scaling --operator deeponet
sbatch --array=0-24 --output=outputs/slurm/gpu_%A_%a.out jobs/slurm/gpu.sh train-scaling --operator fno
# // see: one output file per task; its first lines must say
# //      "Architecture: tuning selection ..." and the chosen layers, activation
# //      and learning rate. Without a finished "lr" stage the job stops with
# //      "No tuning selection" (nothing is trained on a default architecture).
# //      runs  -> outputs/logs/scaling_3200/scale_fno_N0800_seed321/ ...
# //      test errors -> outputs/study_results/scaling_3200/scale_.../

# // ==========================================================================
# // 7. RESULTS OF THE STUDY
# //
# // scaling-report : error against N on log-log axes, fitted alpha
# // compare        : the full DeepONet-FNO report at every N
# // -> outputs/figures/scaling_3200/report/
# // ==========================================================================
sbatch jobs/slurm/cpu.sh scaling-report
for n in 200 400 800 1600 3200
do
    sbatch jobs/slurm/cpu.sh compare --family scaling --train-size $n
done
# // see: scaling-report -> outputs/figures/scaling_3200/report/scaling.pdf,
# //                       scaling.csv, scaling_rates.json (alpha of each operator)
# //      compare        -> outputs/figures/scaling_3200/report/N0200/ ... N3200/

# // ==========================================================================
# // 8. FIGURES AND TIMINGS OF SINGLE RUNS  (example: N = 3200, seed 321)
# // ==========================================================================
R="--family scaling --operator fno --run-name scale_fno_N3200_seed321"
sbatch jobs/slurm/gpu.sh plot $R
sbatch jobs/slurm/gpu.sh evaluate-rounded $R
sbatch jobs/slurm/gpu.sh benchmark $R
# // see: plot      -> outputs/figures/scaling_3200/scale_fno_N3200_seed321/
# //      rounded   -> outputs/study_results/scaling_3200/scale_.../rounded_rect_external_test/
# //      benchmark -> outputs/study_results/scaling_3200/scale_.../benchmarks/


# ##########################################################################
# #  PART C   THE EARLIER 500-CASE STUDY  (no --family option)
# ##########################################################################

# // ==========================================================================
# // 9. THE 500-CASE DATASET  (same code and seed as before = same 500 cases)
# // ==========================================================================
sbatch jobs/slurm/cpu.sh generate --cases 500

# // when it has finished:
sbatch jobs/slurm/cpu.sh generate-rounded --rounded-cases 50

# // ==========================================================================
# // 10. TUNING: block 5 with  T="--operator deeponet"  and  T="--operator fno"
# // ==========================================================================

# // ==========================================================================
# // 11. FINAL MODELS (5 SEEDS), TEST, TIMING, REPORT
# // ==========================================================================
for s in 321 777 2026 31415 27182
do
    sbatch jobs/slurm/gpu.sh train-final --operator deeponet --seed $s
    sbatch jobs/slurm/gpu.sh train-final --operator fno      --seed $s
done

# // when the training jobs have finished:
for s in 321 777 2026 31415 27182
do
    sbatch jobs/slurm/gpu.sh evaluate  --operator deeponet --run-name final_deeponet_seed$s
    sbatch jobs/slurm/gpu.sh evaluate  --operator fno      --run-name final_fno_seed$s
    sbatch jobs/slurm/gpu.sh benchmark --operator deeponet --run-name final_deeponet_seed$s
    sbatch jobs/slurm/gpu.sh benchmark --operator fno      --run-name final_fno_seed$s
done

# // when those have finished:
sbatch jobs/slurm/cpu.sh compare
# // see: outputs/figures/combined_0500/report/   (figures and .csv tables)


# ##########################################################################
# #  PART D   STOPPING, RESULTS
# ##########################################################################

# // ==========================================================================
# // 12. WATCHING JOBS: see block 0
# // ==========================================================================

# // ==========================================================================
# // 13. STOPPED AT THE 24-HOUR LIMIT?
# //
# // tune, train-scaling, generate : submit the same line again
# // train, train-final            : the same line with --resume at the end
# // ==========================================================================

# // ==========================================================================
# // 14. BRING THE RESULTS HOME  (type this on YOUR computer, not on Sunbird)
# // ==========================================================================
scp -r USERNAME@LOGIN_NODE:/scratch/USERNAME/ML/outputs .
