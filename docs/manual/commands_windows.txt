rem ##########################################################################
rem #  PART B   THE FULL STUDY, STEP BY STEP
rem #
rem #  One dataset ("scaling", 3800 FE cases) serves the whole study:
rem #    3200 TRAINING cases from the Halton sequence, in Halton order;
rem #         the training set of size N is the first N of them
rem #         (N = 200, 400, 800, 1600, 3200; each set lies inside the next)
rem #     200 VALIDATION cases  - the same for every N (early stopping, tuning)
rem #     400 TEST cases        - the same for every N (final errors only)
rem #  "Scaling" = how the test error scales with the number of training
rem #  cases N; expected  error = C N^(-alpha)  (a straight line on log-log axes).
rem #  Every command of Part B carries  --family scaling  (or sets it itself).
rem ##########################################################################


rem // ==========================================================================
rem // 7. DATA  (CPU, NGSolve; if a command stops, type it again: it continues)
rem //
rem // generate         : solves the 3800 cases and writes the dataset
rem // generate-rounded : 50 rounded-rectangle cavities, a shape never used in
rem //                    training (extra generalisation test), scaled with the
rem //                    temperature range of the training pool
rem // plot-sampling    : figure of where the cavities lie and their sizes
rem // preview          : one FE case drawn: geometry, mesh, FE field, SDF, mask
rem // mesh-check       : FE accuracy on three meshes + heat balance (table)
rem // ==========================================================================
python run.py generate --family scaling
python run.py generate-rounded --family scaling --rounded-cases 50
python run.py plot-sampling --family scaling
python run.py preview
python run.py mesh-check
rem // results: data\processed\scaling_3200\protocol_v2\dataset.npz
rem //          outputs\figures\scaling_3200\dataset\   (plot-sampling)
rem //          outputs\study_results\combined_0500\preview\ and \mesh_convergence\


rem // ==========================================================================
rem // 8. QUICK TRY OF ONE MODEL  (optional; checks that training works)
rem //
rem //   --train-size 800   train on the first 800 training cases
rem //   --operator fno     which network: fno or deeponet
rem //   --seed 321         random seed (initial weights, order of the batches)
rem //   --run-name ...     name of this run; all its files carry this name
rem //   --epochs 300       short run (the study uses 5000 + early stopping)
rem // One line is printed every 20 epochs; RMSE is the validation error in K.
rem // ==========================================================================
python run.py train    --family scaling --train-size 800 --operator fno --seed 321 --run-name try_fno --epochs 300
python run.py evaluate --family scaling --operator fno --run-name try_fno
python run.py plot     --family scaling --operator fno --run-name try_fno
rem // results: outputs\logs\scaling_3200\try_fno\         loss history
rem //          outputs\study_results\scaling_3200\try_fno\ test errors
rem //          outputs\figures\scaling_3200\try_fno\       figures


rem // ==========================================================================
rem // 9. TUNING AT N = 800  (validation cases only; the test cases are not used)
rem //
rem // tune        : trains every model of one stage, one after another, on the
rem //               first 800 training cases; scores each on the validation set
rem // tune-select : picks the model with the smallest validation error and
rem //               draws the heatmap -> outputs\figures\scaling_3200\report\N0800\
rem // DeepONet : grid = 5 depths x 5 widths, tanh (25 models)
rem //            grid_gelu = the same 25 with GELU
rem //            lr = 3 learning rates at the better of the two grid winners
rem // FNO      : modes_width = 4 x 3 (12), depth = 4 (4), lr = 3 learning rates
rem // Each stage starts from the winner of the previous one: keep the order.
rem // --epochs 2000 gives every tuning model the same budget (fair comparison).
rem // If "tune" stops, type it again: finished models are skipped.
rem // ==========================================================================
python run.py tune        --family scaling --train-size 800 --operator deeponet --stage-name grid --epochs 2000
python run.py tune-select --family scaling --train-size 800 --operator deeponet --stage-name grid
python run.py tune        --family scaling --train-size 800 --operator deeponet --stage-name grid_gelu --epochs 2000
python run.py tune-select --family scaling --train-size 800 --operator deeponet --stage-name grid_gelu
python run.py tune        --family scaling --train-size 800 --operator deeponet --stage-name lr --epochs 2000
python run.py tune-select --family scaling --train-size 800 --operator deeponet --stage-name lr

python run.py tune        --family scaling --train-size 800 --operator fno --stage-name modes_width --epochs 2000
python run.py tune-select --family scaling --train-size 800 --operator fno --stage-name modes_width
python run.py tune        --family scaling --train-size 800 --operator fno --stage-name depth --epochs 2000
python run.py tune-select --family scaling --train-size 800 --operator fno --stage-name depth
python run.py tune        --family scaling --train-size 800 --operator fno --stage-name lr --epochs 2000
python run.py tune-select --family scaling --train-size 800 --operator fno --stage-name lr


rem // ==========================================================================
rem // 10. THE DATA-SIZE RUNS  (5 sizes x 5 seeds = 25 runs per operator)
rem //
rem // train-scaling : for N = 200, 400, 800, 1600, 3200 and seeds 321, 777, 2026, 31415, 27182
rem //                 trains the tuned model on the first N training cases,
rem //                 then scores it on the 400 test cases.
rem //                 Runs are named scale_fno_N0800_seed321, ...
rem //                 Every run: 5000 epochs at most, early stopping after 800
rem //                 epochs without improvement on the validation set.
rem //                 Typing the command again skips finished runs.
rem // ==========================================================================
python run.py train-scaling --operator deeponet
python run.py train-scaling --operator fno
rem // results: outputs\study_results\scaling_3200\scale_fno_N0800_seed321\ ...
rem //          (one folder per run; the window prints which run is training)


rem // ==========================================================================
rem // 11. RESULTS OF THE STUDY
rem //
rem // scaling-report : test error against N on log-log axes, with the fitted
rem //                  rate alpha of every operator
rem //                  -> outputs\figures\scaling_3200\report\scaling.pdf
rem // compare        : the full DeepONet-FNO report at ONE size N (learning
rem //                  curves, test error over seeds, case errors, error per
rem //                  shape, case-by-case scatter), here for every N
rem //                  -> outputs\figures\scaling_3200\report\N0200\ ... \N3200\
rem // for %n in (...) : repeats the command for every size %n
rem // ==========================================================================
python run.py scaling-report
for %n in (200 400 800 1600 3200) do python run.py compare --family scaling --train-size %n


rem // ==========================================================================
rem // 12. FIGURES AND TIMINGS OF SINGLE RUNS  (example: N = 3200, seed 321)
rem //
rem // plot             : loss curve + one test case (FE, prediction, error);
rem //                    --test-case 7 for another case (0 ... 399)
rem // evaluate-rounded : errors on the 50 rounded rectangles
rem // benchmark        : seconds per training epoch and per prediction
rem // The last line plots every one of the 400 test cases of one run.
rem // ==========================================================================
python run.py plot             --family scaling --operator fno --run-name scale_fno_N3200_seed321
python run.py evaluate-rounded --family scaling --operator fno --run-name scale_fno_N3200_seed321
python run.py benchmark        --family scaling --operator fno --run-name scale_fno_N3200_seed321
python run.py plot             --family scaling --operator deeponet --run-name scale_deeponet_N3200_seed321
python run.py evaluate-rounded --family scaling --operator deeponet --run-name scale_deeponet_N3200_seed321
python run.py benchmark        --family scaling --operator deeponet --run-name scale_deeponet_N3200_seed321
for /L %i in (0,1,399) do python run.py plot --family scaling --operator fno --run-name scale_fno_N3200_seed321 --test-case %i
rem // results: outputs\figures\scaling_3200\scale_fno_N3200_seed321\
rem //          outputs\study_results\scaling_3200\scale_fno_N3200_seed321\benchmarks\

rem // ==========================================================================
rem // 14. TUNING ON THE 500-CASE DATASET  (same stages as block 9)
rem // ==========================================================================
python run.py tune        --operator deeponet --stage-name grid --epochs 2000
python run.py tune-select --operator deeponet --stage-name grid
python run.py tune        --operator deeponet --stage-name grid_gelu --epochs 2000
python run.py tune-select --operator deeponet --stage-name grid_gelu
python run.py tune        --operator deeponet --stage-name lr --epochs 2000
python run.py tune-select --operator deeponet --stage-name lr
python run.py tune        --operator fno --stage-name modes_width --epochs 2000
python run.py tune-select --operator fno --stage-name modes_width
python run.py tune        --operator fno --stage-name depth --epochs 2000
python run.py tune-select --operator fno --stage-name depth
python run.py tune        --operator fno --stage-name lr --epochs 2000
python run.py tune-select --operator fno --stage-name lr


rem // ==========================================================================
rem // 15. FINAL MODELS (5 SEEDS), TEST, TIMING AND REPORT
rem //
rem // set S=...           : the five seeds, used as %S%
rem // for %s in (%S%) do  : repeats the command once for every seed %s
rem // train-final         : trains the winner of the last tuning stage;
rem //                       runs are named final_deeponet_seed321, ...
rem // compare             : report -> outputs\figures\combined_0500\report\
rem // ==========================================================================
set S=321 777 2026 31415 27182
for %s in (%S%) do python run.py train-final --operator deeponet --seed %s
for %s in (%S%) do python run.py train-final --operator fno --seed %s
for %s in (%S%) do python run.py evaluate --operator deeponet --run-name final_deeponet_seed%s
for %s in (%S%) do python run.py evaluate --operator fno --run-name final_fno_seed%s
for %s in (%S%) do python run.py benchmark --operator deeponet --run-name final_deeponet_seed%s
for %s in (%S%) do python run.py benchmark --operator fno --run-name final_fno_seed%s
python run.py compare

rem ##########################################################################
